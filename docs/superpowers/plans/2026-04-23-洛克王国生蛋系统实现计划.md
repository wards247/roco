- [ ] **Step 1: 创建 useEggGroups Hook**

```typescript
// src/hooks/useEggGroups.ts
import { useState, useEffect } from 'react';
import { EggGroup, Pokemon } from '../types';
import eggGroupsData from '../data/egg-groups.json';

interface EggGroupData {
  group: EggGroup;
  cards: Pokemon[];
}

export const useEggGroups = () => {
  const [eggGroups, setEggGroups] = useState<EggGroup[]>([]);
  const [allPokemon, setAllPokemon] = useState<Pokemon[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const groups: EggGroup[] = eggGroupsData.eggGroups.map(g => ({
      ...g,
      member_count: g.member_count || 0,
      hatchable_member_count: g.hatchable_member_count || 0
    }));
    setEggGroups(groups);

    const pokemon: Pokemon[] = [];
    Object.entries(eggGroupsData.pokemonByGroup).forEach(([groupId, pokemonList]) => {
      pokemonList.forEach(p => {
        pokemon.push({
          ...p,
          egg_group_id: Number(groupId)
        });
      });
    });
    setAllPokemon(pokemon);
    setLoading(false);
  }, []);

  const getPokemonByGroup = (groupId: number): Pokemon[] => {
    return allPokemon.filter(p => p.egg_group_id === groupId);
  };

  const searchPokemon = (query: string, groupId?: number | null): Pokemon[] => {
    let result = allPokemon;
    if (groupId) {
      result = result.filter(p => p.egg_group_id === groupId);
    }
    if (query.trim()) {
      const q = query.toLowerCase();
      result = result.filter(p =>
        p.display_name.toLowerCase().includes(q) ||
        p.page_name.toLowerCase().includes(q)
      );
    }
    return result;
  };

  return {
    eggGroups,
    allPokemon,
    loading,
    getPokemonByGroup,
    searchPokemon
  };
};
```

### Task 9: 创建 useMyPokemon Hook

**Files:**
- Create: `src/hooks/useMyPokemon.ts`

- [ ] **Step 1: 创建 useMyPokemon Hook**

```typescript
// src/hooks/useMyPokemon.ts
import { useState, useEffect, useCallback } from 'react';
import { MyPokemon } from '../types';

const STORAGE_KEY = 'roco_my_pokemon';

export const useMyPokemon = () => {
  const [myPokemon, setMyPokemon] = useState<MyPokemon[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) {
      try {
        setMyPokemon(JSON.parse(stored));
      } catch {
        setMyPokemon([]);
      }
    }
    setLoading(false);
  }, []);

  const saveToStorage = useCallback((data: MyPokemon[]) => {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
  }, []);

  const addPokemon = useCallback((baseId: number, eggGroupId: number, displayName?: string, avatarUrl?: string) => {
    setMyPokemon(prev => {
      if (prev.some(p => p.base_id === baseId)) return prev;
      const newList = [...prev, {
        base_id: baseId,
        egg_group_id: eggGroupId,
        gender: 'unknown',
        is_mine: true,
        display_name: displayName,
        avatar_url: avatarUrl
      }];
      saveToStorage(newList);
      return newList;
    });
  }, [saveToStorage]);

  const updateGender = useCallback((baseId: number, gender: 'male' | 'female' | 'unknown') => {
    setMyPokemon(prev => {
      const newList = prev.map(p =>
        p.base_id === baseId ? { ...p, gender } : p
      );
      saveToStorage(newList);
      return newList;
    });
  }, [saveToStorage]);

  const removePokemon = useCallback((baseId: number) => {
    setMyPokemon(prev => {
      const newList = prev.filter(p => p.base_id !== baseId);
      saveToStorage(newList);
      return newList;
    });
  }, [saveToStorage]);

  const getMyPokemonMap = useCallback(() => {
    return new Map(myPokemon.map(p => [p.base_id, p]));
  }, [myPokemon]);

  return {
    myPokemon,
    loading,
    addPokemon,
    updateGender,
    removePokemon,
    getMyPokemonMap
  };
};
```

```typescript
// src/hooks/useBreeding.ts
import { useMemo } from 'react';
import { MyPokemon } from '../types';

interface BreedingPair {
  female: MyPokemon;
  compatibleMales: MyPokemon[];
  compatibilityCount: number;
}

export const useBreeding = (myPokemon: MyPokemon[]) => {
  const breedingRecommendations = useMemo(() => {
    const females = myPokemon.filter(p => p.gender === 'female');
    const males = myPokemon.filter(p => p.gender === 'male');

    const recommendations: BreedingPair[] = females.map(female => {
      const compatibleMales = males.filter(male => male.egg_group_id === female.egg_group_id);
      return {
        female,
        compatibleMales,
        compatibilityCount: compatibleMales.length
      };
    });

    recommendations.sort((a, b) => b.compatibilityCount - a.compatibilityCount);

    return recommendations;
  }, [myPokemon]);

  const stats = useMemo(() => {
    const total = myPokemon.length;
    const maleCount = myPokemon.filter(p => p.gender === 'male').length;
    const femaleCount = myPokemon.filter(p => p.gender === 'female').length;
    const unknownCount = myPokemon.filter(p => p.gender === 'unknown').length;

    return { total, maleCount, femaleCount, unknownCount };
  }, [myPokemon]);

  return { breedingRecommendations, stats };
};
```
